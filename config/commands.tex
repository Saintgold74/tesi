% Comandi personalizzati per velocizzare la scrittura

% Figura semplice
\newcommand{\myfig}[4]{
    \begin{figure}[htbp]
        \centering
        \includegraphics[width=#1\textwidth]{figures/#2}
        \caption{#3}
        \label{fig:#4}
    \end{figure}
}

% Tabella semplice
\newcommand{\mytab}[4]{
    \begin{table}[htbp]
        \centering
        \caption{#1}
        \label{tab:#2}
        #3
    \end{table}
}
